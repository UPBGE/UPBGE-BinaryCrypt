# Reference key providers. Standalone (cmake -S keyserver/providers) or included by the
# UPBGE-BinaryCrypt build for its tests. bc_provider_dev and bc_provider_itch (no SDK) always build;
# the other store providers build when their SDK folder is given.

cmake_minimum_required(VERSION 3.16)
project(UPBGE_BinaryCrypt_Providers C)

set(BCP_STEAMWORKS_SDK "" CACHE PATH "Steamworks SDK 'sdk' folder: builds bc_provider_steam")
set(BCP_EOS_SDK "" CACHE PATH "EOS SDK 'SDK' folder: builds bc_provider_epic")
set(BCP_META_PLATFORM_SDK "" CACHE PATH "Meta Platform SDK folder (Include, Windows): builds bc_provider_meta")
set(BCP_KEY_URL "" CACHE STRING "Key server URL baked into the store providers (else the game project's key_url)")

add_library(bcp_common STATIC bc_provider.c)
target_include_directories(bcp_common PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(WIN32)
  target_link_libraries(bcp_common PUBLIC winhttp)
else()
  find_package(CURL REQUIRED)
  target_link_libraries(bcp_common PUBLIC CURL::libcurl)
endif()

add_executable(bc_provider_dev bc_provider_dev.c)
target_link_libraries(bc_provider_dev PRIVATE bcp_common)

add_executable(bc_provider_itch bc_provider_itch.c)
target_link_libraries(bc_provider_itch PRIVATE bcp_common)
if(BCP_KEY_URL)
  target_compile_definitions(bc_provider_itch PRIVATE "BCP_ITCH_KEY_URL=\"${BCP_KEY_URL}\"")
endif()

# Installed into the key server's package (the UPBGE-BinaryCrypt build) or <prefix>/providers
# (standalone), each store provider with the library it loads.
if(NOT BC_PACKAGE_KEYSERVER)
  set(BC_PACKAGE_KEYSERVER .)
endif()
set(bcp_destination ${BC_PACKAGE_KEYSERVER}/providers)
install(TARGETS bc_provider_dev bc_provider_itch RUNTIME DESTINATION ${bcp_destination} COMPONENT keyserver)

if(BCP_STEAMWORKS_SDK)
  enable_language(CXX)
  add_executable(bc_provider_steam bc_provider_steam.cpp)
  target_compile_features(bc_provider_steam PRIVATE cxx_std_11)
  target_include_directories(bc_provider_steam PRIVATE ${BCP_STEAMWORKS_SDK}/public)
  if(WIN32)
    target_link_libraries(bc_provider_steam PRIVATE bcp_common ${BCP_STEAMWORKS_SDK}/redistributable_bin/win64/steam_api64.lib)
    set(bcp_steam_runtime ${BCP_STEAMWORKS_SDK}/redistributable_bin/win64/steam_api64.dll)
  elseif(APPLE)
    set(bcp_steam_runtime ${BCP_STEAMWORKS_SDK}/redistributable_bin/osx/libsteam_api.dylib)
    target_link_libraries(bc_provider_steam PRIVATE bcp_common ${bcp_steam_runtime})
  else()
    set(bcp_steam_runtime ${BCP_STEAMWORKS_SDK}/redistributable_bin/linux64/libsteam_api.so)
    target_link_libraries(bc_provider_steam PRIVATE bcp_common ${bcp_steam_runtime})
    set_target_properties(bc_provider_steam PROPERTIES BUILD_RPATH "$ORIGIN" INSTALL_RPATH "$ORIGIN")
  endif()
  # The library beside the provider, as it ships: bc_tool copies both into the game's provider/.
  add_custom_command(TARGET bc_provider_steam POST_BUILD
                     COMMAND ${CMAKE_COMMAND} -E copy_if_different ${bcp_steam_runtime} $<TARGET_FILE_DIR:bc_provider_steam>)
  install(TARGETS bc_provider_steam RUNTIME DESTINATION ${bcp_destination} COMPONENT keyserver)
  install(FILES ${bcp_steam_runtime} DESTINATION ${bcp_destination} COMPONENT keyserver)
  if(BCP_KEY_URL)
    target_compile_definitions(bc_provider_steam PRIVATE "BCP_STEAM_KEY_URL=\"${BCP_KEY_URL}\"")
  endif()
endif()

if(BCP_EOS_SDK)
  foreach(name PRODUCT_ID SANDBOX_ID DEPLOYMENT_ID CLIENT_ID CLIENT_SECRET CATALOG_ITEM_ID)
    set(BCP_EPIC_${name} "" CACHE STRING "Epic Developer Portal value for bc_provider_epic")
    if(NOT BCP_EPIC_${name})
      message(FATAL_ERROR "bc_provider_epic: set BCP_EPIC_${name}")
    endif()
    list(APPEND bcp_epic_definitions "BCP_EPIC_${name}=\"${BCP_EPIC_${name}}\"")
  endforeach()
  if(BCP_KEY_URL)
    list(APPEND bcp_epic_definitions "BCP_EPIC_KEY_URL=\"${BCP_KEY_URL}\"")
  endif()
  add_executable(bc_provider_epic bc_provider_epic.c)
  target_include_directories(bc_provider_epic PRIVATE ${BCP_EOS_SDK}/Include)
  target_compile_definitions(bc_provider_epic PRIVATE ${bcp_epic_definitions})
  if(WIN32)
    target_link_libraries(bc_provider_epic PRIVATE bcp_common ${BCP_EOS_SDK}/Lib/EOSSDK-Win64-Shipping.lib)
    set(bcp_eos_runtime ${BCP_EOS_SDK}/Bin/EOSSDK-Win64-Shipping.dll)
  elseif(APPLE)
    set(bcp_eos_runtime ${BCP_EOS_SDK}/Bin/libEOSSDK-Mac-Shipping.dylib)
    target_link_libraries(bc_provider_epic PRIVATE bcp_common ${bcp_eos_runtime})
  else()
    set(bcp_eos_runtime ${BCP_EOS_SDK}/Bin/libEOSSDK-Linux-Shipping.so)
    target_link_libraries(bc_provider_epic PRIVATE bcp_common ${bcp_eos_runtime})
    set_target_properties(bc_provider_epic PROPERTIES BUILD_RPATH "$ORIGIN" INSTALL_RPATH "$ORIGIN")
  endif()
  add_custom_command(TARGET bc_provider_epic POST_BUILD
                     COMMAND ${CMAKE_COMMAND} -E copy_if_different ${bcp_eos_runtime} $<TARGET_FILE_DIR:bc_provider_epic>)
  install(TARGETS bc_provider_epic RUNTIME DESTINATION ${bcp_destination} COMPONENT keyserver)
  install(FILES ${bcp_eos_runtime} DESTINATION ${bcp_destination} COMPONENT keyserver)
endif()

if(BCP_META_PLATFORM_SDK)
  if(NOT WIN32)
    message(FATAL_ERROR "bc_provider_meta: the Meta Platform SDK for PC is for Windows only")
  endif()
  set(BCP_META_APP_ID "" CACHE STRING "App id from the Meta Horizon developer dashboard, for bc_provider_meta")
  if(NOT BCP_META_APP_ID)
    message(FATAL_ERROR "bc_provider_meta: set BCP_META_APP_ID")
  endif()
  # The SDK's loader (C++) finds the platform library of the Meta app installed on the computer.
  enable_language(CXX)
  add_executable(bc_provider_meta bc_provider_meta.c ${BCP_META_PLATFORM_SDK}/Windows/OVR_PlatformLoader.cpp)
  target_include_directories(bc_provider_meta PRIVATE ${BCP_META_PLATFORM_SDK}/Include)
  target_compile_definitions(bc_provider_meta PRIVATE "BCP_META_APP_ID=\"${BCP_META_APP_ID}\"")
  if(BCP_KEY_URL)
    target_compile_definitions(bc_provider_meta PRIVATE "BCP_META_KEY_URL=\"${BCP_KEY_URL}\"")
  endif()
  target_link_libraries(bc_provider_meta PRIVATE bcp_common ${BCP_META_PLATFORM_SDK}/Windows/LibOVRPlatform64_1.lib)
  install(TARGETS bc_provider_meta RUNTIME DESTINATION ${bcp_destination} COMPONENT keyserver)
endif()
